let profiles = [];
let tooltip;

function preload() {
  profiles = loadJSON('data.json');
}

function setup() {
  createCanvas(windowWidth, windowHeight);
  noStroke();
  tooltip = document.getElementById("tooltip");
}

function draw() {
  background(10, 10, 20);
  let keys = Object.keys(profiles);
  let spacing = width / (keys.length + 1);

  for (let i = 0; i < keys.length; i++) {
    let p = profiles[keys[i]];
    let percent = p.fieldsFilled / p.totalFields;
    let value = percent * 100;
    let x = spacing * (i + 1);
    let y = height / 2 + sin(frameCount * 0.01 + i) * 30;
    let size = map(value, 0, 100, 60, 200);
    let col = color(255 * percent, 100, 200);
    fill(col);
    drawingContext.shadowBlur = 30;
    drawingContext.shadowColor = col;
    ellipse(x, y, size);

    fill(255);
    textAlign(CENTER);
    textSize(14);
    text(p.id, x, y + size / 2 + 20);

    // hover detection
    let d = dist(mouseX, mouseY, x, y);
    if (d < size / 2) {
      showTooltip(p, x, y, value);
    }
  }
}

function showTooltip(p, x, y, value) {
  tooltip.innerHTML = `
    <b>${p.id}</b><br>
    Market value: $${value.toFixed(2)}<br><br>
    Age: ${p.age}<br>
    From: ${p.from}<br>
    Mood: ${p.mood}<br>
    Heart rate tracked: ${p.heartRate}<br>
    Sleep: ${p.sleep}<br>
    Anxiety: ${p.anxiety}<br>
    Steps: ${p.steps}<br>
    Weight: ${p.weight} kg<br>
    Screen time: ${p.screen} hr<br>
    Workout: ${p.workout}<br>
    Creativity: ${p.creative}<br>
  `;
  tooltip.style.left = (x + 20) + "px";
  tooltip.style.top = (y - 100) + "px";
  tooltip.style.display = "block";
}

function mouseMoved() {
  tooltip.style.display = "none";
}

function windowResized() {
  resizeCanvas(windowWidth, windowHeight);
}
